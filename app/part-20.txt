 if(nr(v,-a))return 'Right size, wrong sign: check whether the answer should be positive or negative.';
 for(var k=1;k<=12;k++)if(nr(v,a*Math.pow(10,k))||nr(v,a/Math.pow(10,k)))return 'Your digits are right, but it\'s out by a factor of 10'+(k>1?'<sup>'+k+'</sup>':'')+'. Check the units and the powers of ten.';
 if(nr(v,1/a))return 'That\'s the answer upside down: check which number divides which.';
 if(nr(v,a*a))return 'Looks like you squared it once too often.';
 if(v>0&&a>0&&nr(v*v,a))return 'That\'s the square root of the answer. Did you forget to square?';
 if(nr(v,a*2)||nr(v,a/2))return 'Out by a factor of 2: look for a ½ or a 2 in the formula.';
 return ''}
